import { useState, useRef, useEffect } from "react";

// ─── Constants ────────────────────────────────────────────────────────────────

const POOL_LABELS = {
  1: "Standard",
  2: "Novice",
  3: "Featured Resonator",
  4: "Featured Weapon",
  5: "Resonator (Alt)",
  6: "Weapon (Alt)",
  7: "Collab",
};
const POOLS = [1, 2, 3, 4, 5, 6, 7];

// Fallback API base when user only has player_id / record_id (overseas)
const FALLBACK_API_BASE = "https://gmserver-api.aki-game2.net/gacha/record/query";

// ─── Utilities ────────────────────────────────────────────────────────────────

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

function parseGachaUrl(raw) {
  try {
    const u = new URL(raw.trim());
    const get = (...keys) => keys.map((k) => u.searchParams.get(k)).find(Boolean) ?? null;
    const playerId = get("playerId", "player_id");
    const recordId = get("recordId", "record_id");
    const svrId    = get("svr_id", "svrId", "svr_area");
    const lang     = get("lang") ?? "en";
    return { playerId, recordId, svrId, lang, valid: !!(playerId && recordId), href: u.href };
  } catch {
    return { valid: false };
  }
}

function toProxyUrl(originalUrl) {
  try {
    const u = new URL(originalUrl);
    // Strip a leading /gacha segment so the fallback URL
    // (whose pathname is already /gacha/record/query) doesn't
    // produce /api/gacha/gacha/record/query.
    const strippedPath = u.pathname.replace(/^\/gacha/, "");
    return `/api/gacha${strippedPath}${u.search}`;
  } catch {
    return null;
  }
}

function downloadJson(data, filename) {
  const blob = new Blob([JSON.stringify(data, null, 2)], { type: "application/json" });
  const a    = Object.assign(document.createElement("a"), {
    href: URL.createObjectURL(blob),
    download: filename,
  });
  a.click();
  URL.revokeObjectURL(a.href);
}

// ─── API helpers ──────────────────────────────────────────────────────────────

async function fetchPoolPulls(baseUrl, cardPoolType, signal) {
  const allPulls    = [];
  let lastRecordId  = "0";
  let hasMore       = true;

  while (hasMore) {
    if (signal?.aborted) throw new DOMException("Aborted", "AbortError");

    const u = new URL(baseUrl);
    u.searchParams.set("cardPoolType", String(cardPoolType));
    u.searchParams.set("recordId",     lastRecordId);

    const proxyUrl = toProxyUrl(u.toString());
    if (!proxyUrl) throw new Error("Invalid URL.");

    let res;
    try {
      res = await fetch(proxyUrl, { signal });
    } catch (err) {
      if (err.name === "AbortError") throw err;
      throw new Error(`Network error: ${err.message}`);
    }

    if (!res.ok) throw new Error(`Server returned HTTP ${res.status}`);

    const json = await res.json();
    if (json?.code !== 0)
      throw new Error(json?.message || `API error (code ${json?.code})`);

    const list = json?.data?.list;
    if (!list || list.length === 0) {
      hasMore = false;
    } else {
      allPulls.push(...list);
      lastRecordId = list[list.length - 1].recordId;
      await sleep(300);
    }
  }
  return allPulls;
}

/** Accepts a File, Blob, or an already-drawn HTMLCanvasElement */
function compressImage(source) {
  const MAX = 1200;

  function scaleCanvas(srcCanvas) {
    let { width, height } = srcCanvas;
    if (width > MAX || height > MAX) {
      const r = Math.min(MAX / width, MAX / height);
      width  = Math.round(width  * r);
      height = Math.round(height * r);
    }
    const out = document.createElement("canvas");
    out.width  = width;
    out.height = height;
    out.getContext("2d").drawImage(srcCanvas, 0, 0, width, height);
    return out.toDataURL("image/jpeg", 0.85).split(",")[1];
  }

  // Already a canvas — compress directly
  if (source instanceof HTMLCanvasElement) {
    return Promise.resolve(scaleCanvas(source));
  }

  // File / Blob — load via Image first
  return new Promise((resolve, reject) => {
    const img    = new Image();
    const objUrl = URL.createObjectURL(source);
    img.onload = () => {
      const tmp    = document.createElement("canvas");
      tmp.width    = img.width;
      tmp.height   = img.height;
      tmp.getContext("2d").drawImage(img, 0, 0);
      URL.revokeObjectURL(objUrl);
      resolve(scaleCanvas(tmp));
    };
    img.onerror = () => { URL.revokeObjectURL(objUrl); reject(new Error("Failed to load image.")); };
    img.src = objUrl;
  });
}

async function extractIdsFromImage(base64Image) {
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model:      "claude-sonnet-4-20250514",
      max_tokens: 300,
      messages: [{
        role: "user",
        content: [
          {
            type:   "image",
            source: { type: "base64", media_type: "image/jpeg", data: base64Image },
          },
          {
            type: "text",
            text: `This is a Wuthering Waves gacha/convene history URL screenshot.
Extract player_id (or playerId), record_id (or recordId), and svr_id (or svrId / svr_area) from the URL.
Respond ONLY with valid JSON — no markdown, no explanation:
{"player_id":"VALUE_OR_NULL","record_id":"VALUE_OR_NULL","svr_id":"VALUE_OR_NULL"}`,
          },
        ],
      }],
    }),
  });

  if (!res.ok) {
    const err = await res.json().catch(() => ({}));
    throw new Error(err?.error?.message || `API error ${res.status}`);
  }

  const data = await res.json();
  const text = data.content?.find((b) => b.type === "text")?.text ?? "";
  try {
    return JSON.parse(text.replace(/```json|```/g, "").trim());
  } catch {
    throw new Error("Could not parse OCR response — try a clearer screenshot.");
  }
}

// ─── Main component ───────────────────────────────────────────────────────────

export default function WuwaImporter({ onPullsFetched }) {
  // ── ID state ──
  const [rawUrl,   setRawUrl]   = useState("");
  const [playerId, setPlayerId] = useState("");
  const [recordId, setRecordId] = useState("");
  const [svrId,    setSvrId]    = useState("");
  const [parsed,   setParsed]   = useState(null);

  // ── Scanner ──
  const [scanStatus, setScanStatus] = useState("idle"); // idle|scanning|done|error
  const [preview,    setPreview]    = useState(null);
  const [scanError,  setScanError]  = useState("");
  const [cameraOpen, setCameraOpen] = useState(false);

  // ── Fetch ──
  const [fetchStatus,   setFetchStatus]   = useState("idle"); // idle|fetching|done|error
  const [poolProgress,  setPoolProgress]  = useState({});     // { poolType: {status, count} }
  const [currentPool,   setCurrentPool]   = useState(null);
  const [fetchError,    setFetchError]    = useState("");
  const [result,        setResult]        = useState(null);

  const fileRef    = useRef(null);
  const abortRef   = useRef(null);
  const previewRef = useRef(null); // tracks the current object URL, or null for data: previews
  const ocrGenRef  = useRef(0);    // incremented on each OCR call and on reset; stale callbacks bail out

  useEffect(() => () => { if (previewRef.current) URL.revokeObjectURL(previewRef.current); }, []);

  // ── URL parsing ──
  function handleUrlChange(e) {
    const val = e.target.value;
    setRawUrl(val);
    setFetchError("");
    const p = parseGachaUrl(val);
    if (p.valid) {
      setParsed(p);
      if (p.playerId) setPlayerId(p.playerId);
      if (p.recordId) setRecordId(p.recordId);
      if (p.svrId)    setSvrId(p.svrId);
    } else {
      setParsed(null);
    }
  }

  // ── Scanner: unified preview setter ──
  // source is either a File/Blob (File extends Blob, so instanceof Blob catches both)
  // → creates a tracked object URL that previewRef can revoke later.
  // Or a data: string (canvas toDataURL) → stored as-is, previewRef stays null
  // because there is no object URL to revoke.
  function _setPreview(source) {
    if (previewRef.current) {
      URL.revokeObjectURL(previewRef.current);
      previewRef.current = null;
    }
    if (source instanceof Blob) {
      const objUrl = URL.createObjectURL(source);
      previewRef.current = objUrl;
      setPreview(objUrl);
    } else {
      // data: string — previewRef stays null; resetScanner won't try to revoke it
      setPreview(source);
    }
  }

  // ── Scanner ── (file drop / file picker in the main dropzone)
  async function processImage(file) {
    if (!file?.type.startsWith("image/")) return;
    _setPreview(file);
    setCameraOpen(false);
    await _runOcr(await compressImage(file));
  }

  // ── Scanner ── (camera: canvas from live viewfinder, or File from fallback picker)
  async function processCanvas(canvasOrFile) {
    if (canvasOrFile instanceof File) {
      // Fallback picker: createObjectURL synchronously before onClose unmounts the input,
      // so the browser can't revoke file access mid-flight.
      _setPreview(canvasOrFile);
      await _runOcr(await compressImage(canvasOrFile));
    } else {
      // Canvas snapshot: use a data URL (no object URL to track or revoke).
      _setPreview(canvasOrFile.toDataURL("image/jpeg", 0.85));
      await _runOcr(await compressImage(canvasOrFile));
    }
  }

  async function _runOcr(base64) {
    // Capture the current generation. If resetScanner increments ocrGenRef while we
    // await, the stale callback detects the mismatch and discards its result.
    ocrGenRef.current += 1;
    const gen = ocrGenRef.current;

    setScanStatus("scanning");
    setScanError("");
    try {
      const ids = await extractIdsFromImage(base64);
      if (ocrGenRef.current !== gen) return; // superseded or reset
      if (!ids.player_id && !ids.record_id)
        throw new Error("IDs not found — try a clearer screenshot.");
      if (ids.player_id) setPlayerId(ids.player_id);
      if (ids.record_id) setRecordId(ids.record_id);
      if (ids.svr_id)    setSvrId(ids.svr_id);
      setScanStatus("done");
    } catch (err) {
      if (ocrGenRef.current !== gen) return;
      setScanStatus("error");
      setScanError(err.message ?? "Scan failed.");
    }
  }

  function resetScanner() {
    ocrGenRef.current += 1; // invalidate any in-flight OCR
    setScanStatus("idle");
    setCameraOpen(false);
    if (previewRef.current) { URL.revokeObjectURL(previewRef.current); previewRef.current = null; }
    setPreview(null);
    setScanError("");
    if (fileRef.current) fileRef.current.value = "";
  }

  // ── Fetcher ──
  async function handleFetch() {
    const pid = playerId.trim();
    const rid = recordId.trim();
    if (!pid || !rid) { setFetchError("player_id and record_id are required."); return; }

    let baseUrl;
    try {
      const usingFallback = !rawUrl.trim();
      const src = usingFallback ? FALLBACK_API_BASE : rawUrl.trim();
      const u = new URL(src);
      // The fallback overseas API uses snake_case; the in-game URL uses camelCase.
      if (usingFallback) {
        u.searchParams.set("player_id", pid);
        u.searchParams.set("record_id", rid);
        if (svrId.trim()) u.searchParams.set("svr_id", svrId.trim());
      } else {
        u.searchParams.set("playerId", pid);
        u.searchParams.set("recordId", rid);
        if (svrId.trim()) u.searchParams.set("svr_id", svrId.trim());
      }
      baseUrl = u.toString();
    } catch { setFetchError("Invalid URL."); return; }

    abortRef.current = new AbortController();
    setFetchStatus("fetching");
    setFetchError("");
    setPoolProgress({});
    setResult(null);

    const allPulls = {};
    let   total    = 0;

    try {
      for (const poolType of POOLS) {
        if (abortRef.current.signal.aborted) break;
        setCurrentPool(poolType);
        setPoolProgress((p) => ({ ...p, [poolType]: { status: "fetching", count: 0 } }));
        try {
          const pulls = await fetchPoolPulls(baseUrl, poolType, abortRef.current.signal);
          if (pulls.length > 0) { allPulls[POOL_LABELS[poolType]] = pulls; total += pulls.length; }
          setPoolProgress((p) => ({ ...p, [poolType]: { status: "done", count: pulls.length } }));
        } catch (err) {
          if (err.name === "AbortError") throw err;
          setPoolProgress((p) => ({ ...p, [poolType]: { status: "error", count: 0 } }));
        }
      }

      if (abortRef.current?.signal.aborted) { setFetchStatus("idle"); setCurrentPool(null); return; }

      const resultData = {
        pulls: allPulls, playerId: pid, recordId: rid, total,
        fetchedAt: new Date().toISOString(),
      };
      setResult(resultData);
      setFetchStatus("done");
      setCurrentPool(null);
      onPullsFetched?.(resultData);
    } catch (err) {
      if (err.name === "AbortError") {
        setFetchStatus("idle"); setCurrentPool(null);
      } else {
        setFetchStatus("error"); setCurrentPool(null);
        setFetchError(err.message ?? "Fetch failed. URL may have expired — reopen Convene History and try again.");
      }
    }
  }

  function handleCancel() { abortRef.current?.abort(); }

  function handleExport() {
    if (!result) return;
    downloadJson(result, `wuwa-pulls-${result.playerId}-${Date.now()}.json`);
  }

  // ── Derived ──
  const completedPools = Object.values(poolProgress).filter((p) => p.status === "done" || p.status === "error").length;
  const progressPct    = Math.round((completedPools / POOLS.length) * 100);
  const totalSoFar     = Object.values(poolProgress).reduce((s, p) => s + (p.count ?? 0), 0);
  const canFetch       = playerId.trim() && recordId.trim() && fetchStatus !== "fetching";

  // ── Render ──
  return (
    <div style={S.card}>
      {/* Scanline overlay */}
      <div style={S.scanlines} aria-hidden="true" />

      {/* Header */}
      <div style={S.headerRow}>
        <TideIcon />
        <div>
          <h2 style={S.title}>Convene Importer</h2>
          <p style={S.subtitle}>WUTHERING WAVES · GACHA HISTORY</p>
        </div>
      </div>

      {/* ── Step 1 ── */}
      <StepLabel n={1} text="Get your IDs" />

      {scanStatus === "idle" && !cameraOpen && (
        <div
          className="wuwa-dropzone"
          style={S.dropzone}
          onDragOver={(e) => e.preventDefault()}
          onDrop={(e) => { e.preventDefault(); processImage(e.dataTransfer.files[0]); }}
        >
          <div style={S.dropIconWrap}><span style={S.dropEmoji}>◎</span></div>
          <p style={S.dropText}>Drop URL screenshot here</p>
          <p style={S.dropHint}>Claude Vision auto-extracts your IDs</p>
          <div style={S.btnRow}>
            <button style={S.btnTeal} onClick={() => setCameraOpen(true)}>📸 Camera</button>
            <button className="wuwa-btn-ghost" style={S.btnGhost} onClick={() => fileRef.current.click()}>🖼 File</button>
          </div>
        </div>
      )}

      {cameraOpen && (
        <CameraScanner
          onCapture={processCanvas}
          onClose={() => setCameraOpen(false)}
        />
      )}

      {scanStatus === "scanning" && (
        <div style={S.previewBox}>
          {preview && <img src={preview} alt="preview" style={S.previewImg} />}
          <div style={S.previewOverlay}>
            <Spinner /> <span style={S.scanningText}>Scanning…</span>
          </div>
        </div>
      )}

      {(scanStatus === "done" || scanStatus === "error") && (
        <div style={S.previewBox}>
          {preview && <img src={preview} alt="preview" style={{ ...S.previewImg, opacity: 0.3, filter: "blur(2px)" }} />}
          <div style={S.previewOverlay}>
            {scanStatus === "done"
              ? <span style={S.okMsg}>✓ IDs extracted successfully</span>
              : <span style={S.errMsg}>✗ {scanError}</span>}
            <button style={S.scanAgainBtn} onClick={resetScanner}>↺ Scan again</button>
          </div>
        </div>
      )}

      <input ref={fileRef} type="file" accept="image/*"
        onChange={(e) => processImage(e.target.files[0])} style={{ display: "none" }} />

      {/* Divider */}
      <div style={S.divider}><span style={S.dividerLine} /><span style={S.dividerText}>or paste / enter manually</span><span style={S.dividerLine} /></div>

      {/* URL field */}
      <label style={S.label}>Convene URL <span style={S.labelNote}>(auto-fills IDs below)</span></label>
      <textarea
        className="wuwa-input"
        style={S.textarea}
        placeholder="https://gmserver-api.aki-game2.net/gacha/record/query?playerId=…"
        value={rawUrl}
        onChange={handleUrlChange}
        rows={3}
        spellCheck={false}
      />
      {parsed && (
        <div style={S.parsedBadge}>
          <span style={S.pulseDot} /> URL parsed — <Mono>{parsed.playerId}</Mono> / <Mono>{parsed.recordId}</Mono>
        </div>
      )}

      {/* ID row */}
      <div style={S.row}>
        <IdField label="player_id" value={playerId} onChange={(e) => setPlayerId(e.target.value)} placeholder="e.g. 500123456" />
        <IdField label="record_id" value={recordId} onChange={(e) => setRecordId(e.target.value)} placeholder="alphanumeric key" />
      </div>

      {/* svr_id */}
      <div style={{ marginBottom: 4 }}>
        <IdField
          label={<>svr_id <span style={S.labelNote}>(optional)</span></>}
          value={svrId}
          onChange={(e) => setSvrId(e.target.value)}
          placeholder="e.g. 76 · overseas01"
        />
      </div>

      {/* ── Step 2 ── */}
      <StepLabel n={2} text="Import" style={{ marginTop: 26 }} />

      {fetchStatus === "fetching" ? (
        <FetchingView
          poolProgress={poolProgress}
          currentPool={currentPool}
          progressPct={progressPct}
          totalSoFar={totalSoFar}
          onCancel={handleCancel}
        />
      ) : (
        <>
          <button
            className="wuwa-btn-import"
            style={{ ...S.importBtn, opacity: canFetch ? 1 : 0.4, cursor: canFetch ? "pointer" : "not-allowed" }}
            onClick={handleFetch}
            disabled={!canFetch}
          >
            ↓ Import Convene History
          </button>

          {fetchStatus === "done" && result && (
            <DoneView result={result} poolProgress={poolProgress} onExport={handleExport} />
          )}

          {fetchError && <p style={S.errMsg}>✗ {fetchError}</p>}
        </>
      )}
    </div>
  );
}

// ─── Sub-components ───────────────────────────────────────────────────────────

function StepLabel({ n, text, style: extra }) {
  return (
    <div style={{ ...S.stepLabel, ...extra }}>
      <span style={S.stepNum}>{n}</span>
      <span style={S.stepText}>{text}</span>
      <span style={S.stepLine} />
    </div>
  );
}

function IdField({ label, value, onChange, placeholder }) {
  return (
    <div style={{ flex: 1 }}>
      <label style={S.label}>{label}</label>
      <input className="wuwa-input" style={S.input} value={value} onChange={onChange} placeholder={placeholder} spellCheck={false} />
    </div>
  );
}

function FetchingView({ poolProgress, currentPool, progressPct, totalSoFar, onCancel }) {
  return (
    <div style={{ animation: "wuwa-slidein .3s ease" }}>
      {/* Progress bar */}
      <div style={S.progressTrack}>
        <div style={{ ...S.progressFill, width: `${progressPct}%` }} />
      </div>
      <p style={S.progressLabel}>{progressPct}% · {totalSoFar} pulls retrieved</p>

      {/* Pool chips */}
      <div style={S.poolGrid}>
        {POOLS.map((pt) => {
          const info = poolProgress[pt];
          const isActive = currentPool === pt;
          return (
            <div
              key={pt}
              style={{
                ...S.chip,
                ...(isActive              ? S.chipActive : {}),
                ...(info?.status === "done"  ? S.chipDone   : {}),
                ...(info?.status === "error" ? S.chipErr    : {}),
              }}
            >
              <span style={S.chipLabel}>{POOL_LABELS[pt]}</span>
              <span style={S.chipRight}>
                {isActive                    && <Spinner small />}
                {info?.status === "done"     && <span style={{ color: "#2dd4bf" }}>{info.count}</span>}
                {info?.status === "error"    && <span style={{ color: "#fb7185", fontSize: 9 }}>err</span>}
              </span>
            </div>
          );
        })}
      </div>

      <button style={S.cancelBtn} onClick={onCancel}>✕ Cancel import</button>
    </div>
  );
}

function DoneView({ result, poolProgress, onExport }) {
  return (
    <div style={S.doneCard}>
      <div style={S.doneBadge}>✓ Complete</div>
      <p style={S.doneStat}><Mono>{result.total}</Mono> pulls · <Mono>{Object.keys(result.pulls).length}</Mono> banners</p>

      {/* Per-banner breakdown */}
      <div style={S.breakdown}>
        {Object.entries(poolProgress)
          .filter(([, info]) => info.count > 0)
          .map(([pt, info]) => (
            <div key={pt} style={S.breakdownRow}>
              <span style={S.breakdownLabel}>{POOL_LABELS[pt]}</span>
              <span style={S.breakdownCount}>{info.count}</span>
            </div>
          ))}
      </div>

      <button style={S.exportBtn} onClick={onExport}>⬇ Export JSON</button>
    </div>
  );
}

function TideIcon() {
  return (

// ─── CameraScanner ───────────────────────────────────────────────────────────
// Live viewfinder using getUserMedia. Falls back to native file picker on error.

function CameraScanner({ onCapture, onClose }) {
  const videoRef     = useRef(null);
  const streamRef    = useRef(null);
  const trackRef     = useRef(null);
  const [ready,      setReady]      = useState(false);
  const [torchOn,    setTorchOn]    = useState(false);
  const [torchAvail, setTorchAvail] = useState(false);
  const [camError,   setCamError]   = useState("");
  const [flash,      setFlash]      = useState(false);
  const fallbackRef  = useRef(null);

  // Start stream
  useEffect(() => {
    let cancelled = false;
    async function start() {
      try {
        const stream = await navigator.mediaDevices.getUserMedia({
          video: { facingMode: { ideal: "environment" }, width: { ideal: 1920 }, height: { ideal: 1080 } },
          audio: false,
        });
        if (cancelled) { stream.getTracks().forEach((t) => t.stop()); return; }
        streamRef.current = stream;
        const track = stream.getVideoTracks()[0];
        trackRef.current = track;
        if (videoRef.current) {
          videoRef.current.srcObject = stream;
          await videoRef.current.play();
        }
        // Detect torch support
        const caps = track.getCapabilities?.() ?? {};
        setTorchAvail(!!(caps.torch));
        setReady(true);
      } catch (err) {
        if (!cancelled) setCamError(err.name === "NotAllowedError"
          ? "Camera access denied — please allow camera in browser settings."
          : `Camera unavailable: ${err.message}`);
      }
    }
    start();
    return () => {
      cancelled = true;
      streamRef.current?.getTracks().forEach((t) => t.stop());
    };
  }, []);

  // Toggle torch
  async function toggleTorch() {
    if (!trackRef.current || !torchAvail) return;
    const next = !torchOn;
    try {
      await trackRef.current.applyConstraints({ advanced: [{ torch: next }] });
      setTorchOn(next);
    } catch { /* ignore */ }
  }

  // Capture a frame
  function capture() {
    const video = videoRef.current;
    if (!video || !ready) return;
    const canvas = document.createElement("canvas");
    canvas.width  = video.videoWidth;
    canvas.height = video.videoHeight;
    canvas.getContext("2d").drawImage(video, 0, 0);
    // Flash effect
    setFlash(true);
    setTimeout(() => setFlash(false), 600);
    // Stop the stream immediately so the camera indicator light turns off at
    // the moment of capture. The cleanup effect calls .stop() again on unmount,
    // which is a no-op per the MediaStreamTrack spec for already-ended tracks.
    streamRef.current?.getTracks().forEach((t) => t.stop());
    onCapture(canvas);
  }

  // Fallback: open native file picker
  function openFileFallback() {
    fallbackRef.current?.click();
  }

  return (
    <div style={CS.wrap}>
      {/* Shutter flash */}
      {flash && <div style={CS.shutterFlash} />}

      {camError ? (
        <div style={CS.errBox}>
          <p style={CS.errText}>⚠ {camError}</p>
          <div style={{ display: "flex", gap: 10, justifyContent: "center", flexWrap: "wrap" }}>
            <button style={CS.fbBtn} onClick={openFileFallback}>🖼 Upload screenshot instead</button>
            <button style={CS.closeBtn} className="wuwa-cam-close" onClick={onClose}>✕ Cancel</button>
          </div>
          <input
            ref={fallbackRef}
            type="file"
            accept="image/*"
            capture="environment"
            onChange={(e) => {
              const file = e.target.files?.[0];
              if (!file) return;
              // createObjectURL (called synchronously inside onCapture → _setPreview)
              // must happen before onClose unmounts this input and potentially
              // invalidates file access on some browsers.
              onCapture(file);
              onClose();
            }}
            style={{ display: "none" }}
          />
        </div>
      ) : (
        <>
          {/* Video */}
          <video
            ref={videoRef}
            muted
            playsInline
            style={{ ...CS.video, opacity: ready ? 1 : 0 }}
          />

          {/* Loading */}
          {!ready && (
            <div style={CS.loadingOverlay}>
              <Spinner /><span style={{ color: "#5eead4", fontSize: 12, fontFamily: "'IBM Plex Mono',monospace" }}>Starting camera…</span>
            </div>
          )}

          {/* Targeting overlay */}
          {ready && (
            <div style={CS.overlay} aria-hidden="true">
              {/* Dimmed regions around the target box */}
              <div style={{ ...CS.dim, top: 0,    left: 0,   right: 0,    height: "28%" }} />
              <div style={{ ...CS.dim, bottom: 0, left: 0,   right: 0,    height: "28%" }} />
              <div style={{ ...CS.dim, top: "28%",left: 0,   width: "5%", height: "44%" }} />
              <div style={{ ...CS.dim, top: "28%",right: 0,  width: "5%", height: "44%" }} />

              {/* Target box corners */}
              <div style={{ ...CS.corner, top: "28%",    left:  "5%",  borderTopWidth:    2, borderLeftWidth:  2 }} />
              <div style={{ ...CS.corner, top: "28%",    right: "5%",  borderTopWidth:    2, borderRightWidth: 2 }} />
              <div style={{ ...CS.corner, bottom: "28%", left:  "5%",  borderBottomWidth: 2, borderLeftWidth:  2 }} />
              <div style={{ ...CS.corner, bottom: "28%", right: "5%",  borderBottomWidth: 2, borderRightWidth: 2 }} />

              {/* Scan line — container spans the target zone so translateY(100%)
                  travels exactly from top to bottom of the scanned area */}
              <div style={CS.scanLineTrack}>
                <div style={CS.scanLine} />
              </div>

              {/* Guide text */}
              <p style={CS.guideText}>Aim at the URL dialog</p>
            </div>
          )}

          {/* Controls bar */}
          {ready && (
            <div style={CS.controls}>
              <button className="wuwa-cam-close" style={CS.iconBtn} onClick={onClose} title="Cancel">✕</button>

              <button className="wuwa-cap-btn" style={CS.capBtn} onClick={capture} title="Capture">
                <span style={CS.capRing} />
                <span style={CS.capDot} />
              </button>

              {torchAvail
                ? <button className="wuwa-torch-btn" style={{ ...CS.iconBtn, color: torchOn ? "#fde68a" : "#3a5568" }} onClick={toggleTorch} title="Torch">⚡</button>
                : <span style={CS.iconBtn} />}
            </div>
          )}
        </>
      )}
    </div>
  );
}
